Архитектура

Два Python-модуля и один общий сетевой слой. Krim_logs.py отвечает за Discord и бизнес-логику, netbypass.py — за транспорт до Discord.

Сетевой поток

discord.py→ client.http.proxy→ netbypass→ VLESS / HTTPS proxy→ Discord

discord.py 2.x передаёт прокси и в REST-запросы, и в WebSocket-шлюз, поэтому одного client.http.proxy достаточно для обоих каналов.

Поток ИИ

message→ handle_ai / handle_deepneuro→ attachments→ call_ai→ reply

Обычный чат использует model. DeepNeuro сначала зовёт deep_mini_model (Krim DeepNeuro Mini) за черновиком, затем deep_model (Krim DeepNeuro) проходит до 4 шагов.

Поток логов

Discord event→ send_log→ log_channel→ Embed/текст

События обрабатываются в on_message, on_message_edit, on_message_delete, on_voice_state_update и других обработчиках.

Секреты (DISCORD_TOKEN, AI_API_KEY, ADMIN_IDS) хранятся в .env и не включены в этот HTML.

Безопасность

Права администратора

Список ADMIN_IDS из .env. Команды log!proxy и log!deepneuro находятся после админской проверки в handle_command.

Только HTTPS-прокси

Каждый кандидат проверяется двумя последовательными CONNECT-запросами к Discord с валидацией TLS. HTTP-прокси и MITM-прокси отбрасываются.

Локальный Xray

VLESS-инбаунд Xray слушает только 127.0.0.1. Наружу не открывается порт и не требуется системный VPN-клиент.

Ограничения файлов

До 10 вложений, до 5 МБ на файл, 6000 символов на файл и 15000 символов суммарно. Бинарные файлы пропускаются.

Секреты

Токены и ключи не попадают в HTML-презентацию. В ней только исходный код и примеры конфигурации без значений секретов.

Контекст ИИ

История ограничена context_limit, хранится в SQLite и может быть сброшена командой log!dialog-clear.

def _deep_merge(base: dict, override: dict) -> dict:
    """Рекурсивно накладывает пользовательский конфиг на дефолтный."""
    result = dict(base)
    for key, value in override.items():
        if isinstance(value, dict) and isinstance(result.get(key), dict):
            result[key] = _deep_merge(result[key], value)
        else:
            result[key] = value
    return result

def load_config() -> None:
    global log_channel_id, emperial_id, guilds_config, voice_guild_ids, voice_auto_enabled, network_config

        return

    if not isinstance(data, dict):
        return

    if isinstance(data.get("log_channel_id"), int):
        log_channel_id = data["log_channel_id"]
    if isinstance(data.get("emperial_id"), int):
        emperial_id = data["emperial_id"]

        guilds_config = data["guilds"]

    need_save = False
    raw_voice = data.get("voice_guilds")
    if isinstance(raw_voice, list):
        parsed: set[int] = set()
        for x in raw_voice:
            try:
                parsed.add(int(x))
            except (TypeError, ValueError):
                continue
        voice_guild_ids = parsed
    else:
        need_save = True

    if isinstance(data.get("voice_auto"), bool):
        voice_auto_enabled = data["voice_auto"]
    else:
        need_save = True

    raw_network = data.get("network")
    if isinstance(raw_network, dict):
        network_config = _deep_merge(copy.deepcopy(DEFAULT_NETWORK_CONFIG), raw_network)
    else:
        need_save = True

    if need_save:  # дозаписываем новые поля, чтобы их было видно в config.json
        save_config()

async def db_execute(sql: str, params: tuple = ()):
    def _run():
        conn = sqlite3.connect(DB_PATH)
        conn.row_factory = sqlite3.Row
        try:
            cur = conn.execute(sql, params)
            head = sql.strip().upper()
            if head.startswith("SELECT") or head.startswith("WITH"):
                return [dict(r) for r in cur.fetchall()]
            conn.commit()
            return None
        finally:
            conn.close()

    try:
        return await asyncio.to_thread(_run)
    except sqlite3.OperationalError as exc:
        # Схема могла быть удалена или БД создана впервые — поднимаем её и повторяем.
        if "no such table" in str(exc).lower():
            log.warning("Схема БД отсутствует — создаю таблицы заново (%s)", exc)
            init_db()
            return await asyncio.to_thread(_run)
        raise

VOSK_RATE = 16000

def pcm_to_vosk(pcm: bytes, in_rate: int = 48000) -> bytes:
    """Конвертирует PCM 48кГц Stereo s16le (от Discord) в 16кГц Mono s16le (для Vosk)."""
    if not pcm:
        return b""
    try:
        # Переводим байты в массив numpy (16-битные целые)
        audio_array = np.frombuffer(pcm, dtype=np.int16)
        
        # Если количество сэмплов нечетное (битый пакет) — отрезаем хвост
        if len(audio_array) % 2 != 0:
            audio_array = audio_array[:-1]
            
        # Стерео -> Моно (усредняем левый и правый каналы)
        audio_array = audio_array.reshape(-1, 2).mean(axis=1).astype(np.int16)
        
        # 48000 Гц -> 16000 Гц (берем каждый 3-й сэмпл)
        # Для распознавания речи это работает отлично
        if in_rate == 48000:
            audio_array = audio_array[::3]
            
        return audio_array.tobytes()
    except Exception as e:
        log.error("Ошибка конвертации аудио: %s", e)
        return b""


class TranscriptionSink(voice_recv.AudioSink):
    def __init__(self):
        super().__init__()
        self.recognizers: dict[int, vosk.KaldiRecognizer] = {}

    def wants_opus(self) -> bool:
        return False

    def cleanup(self) -> None:
        self.recognizers.clear()

    def write(self, user: Optional[discord.User], data: voice_recv.VoiceData):
        if not user or not vosk_model:
            return
        if client.user and user.id == client.user.id:
            return

        # ОТЛАДКА: выводим в консоль, что пакет от пользователя получен
        # (Отключи этот print, когда убедишься, что всё работает)
        # log.debug("Получен аудио-пакет от %s", user.display_name)

        try:
            pcm = pcm_to_vosk(data.pcm)
            if not pcm:
                return
                
            rec = self.recognizers.get(user.id)
            if rec is None:
                rec = self.recognizers[user.id] = vosk.KaldiRecognizer(vosk_model, VOSK_RATE)
            
            # AcceptWaveform возвращает True, когда Vosk накапливает достаточно звука и слышит паузу
            if rec.AcceptWaveform(pcm):
                text = (json.loads(rec.Result()).get("text") or "").strip()
                if text:
                    log.info("Vosk РАСПОЗНАЛ [%s]: %s", user.display_name, text)
                    asyncio.run_coroutine_threadsafe(
                        self.process_text(user, text), client.loop
                    )
        except Exception:
            log.exception("Ошибка распознавания речи (%s)", getattr(user, "id", "?"))
            
    # process_text оставляем без изменений
    async def process_text(self, user: discord.User, text: str) -> None:
        # data.source — это пользователь, а не канал: канал берём у voice-клиента
        channel = getattr(self.voice_client, "channel", None)
        if channel is None:
            return
        try:
            avatar = getattr(getattr(user, "display_avatar", None), "url", "")
            await db_execute(
                "INSERT INTO voice_transcripts (guild_id, channel_id, channel_name, user_id, user_name, user_avatar, text, timestamp) "
                "VALUES (?, ?, ?, ?, ?, ?, ?, ?)",
                (
                    channel.guild.id,
                    channel.id,
                    channel.name,
                    user.id,
                    getattr(user, "display_name", str(user)),
                    avatar,
                    text,
                    time.time(),
                ),
            )
            await generate_transcript_html()
        except Exception:
            log.exception("Ошибка сохранения транскрипта (%s)", user.id)

def parse_duration(dur_str: str) -> int:
    m = re.match(r"^(\d+)([mhd])$", dur_str.lower())
    if not m: return 0
    val, unit = int(m.group(1)), m.group(2)
    if unit == 'm': return val * 60
    if unit == 'h': return val * 3600
    if unit == 'd': return val * 86400
    return 0

async def apply_mute(guild: discord.Guild, member: discord.Member, duration_sec: int, reason: str = "Krim Logs"):
    cfg = guilds_config.get(str(guild.id))
    if not cfg: return False, "Сервер не настроен в config.json для мутов."

    unmute_at = time.time() + duration_sec
    mode = cfg.get("mode", "role")
    
    if mode == "role":
        role_id = cfg.get("mute_role_id")
        if not role_id: return False, "Не указан ID роли для мутов."
        role = guild.get_role(role_id)
        if not role: return False, "Роль для мутов не найдена."
        try:
            await member.add_roles(role, reason=reason)
        except discord.Forbidden:
            return False, "У бота нет прав на выдачу роли (или роль бота ниже)."
    elif mode == "permissions":
        try:
            await member.timeout(timedelta(seconds=duration_sec), reason=reason)
        except discord.Forbidden:
            return False, "У бота нет прав на тайм-аут участников."

    await db_execute("INSERT INTO active_mutes(guild_id, user_id, unmute_at) VALUES (?, ?, ?) ON CONFLICT(guild_id, user_id) DO UPDATE SET unmute_at=excluded.unmute_at", (guild.id, member.id, unmute_at))
    asyncio.create_task(unmute_later(guild, member.id, duration_sec))
    return True, "OK"

async def remove_mute(guild: discord.Guild, member: discord.Member, reason: str = "Krim Logs Unmute"):
    cfg = guilds_config.get(str(guild.id))
    if not cfg: return False, "Сервер не настроен в config.json."

    mode = cfg.get("mode", "role")
    if mode == "role":
        role_id = cfg.get("mute_role_id")
        if role_id:
            role = guild.get_role(role_id)
            if role and role in member.roles:
                try: await member.remove_roles(role, reason=reason)
                except discord.Forbidden: return False, "Нет прав на снятие роли."
    elif mode == "permissions":
        try: await member.timeout(None, reason=reason)
        except discord.Forbidden: return False, "Нет прав на снятие тайм-аута."

    await db_execute("DELETE FROM active_mutes WHERE guild_id=? AND user_id=?", (guild.id, member.id))
    return True, "OK"

async def unmute_later(guild: discord.Guild, user_id: int, delay: int):
    await asyncio.sleep(delay)
    member = guild.get_member(user_id)
    if member: await remove_mute(guild, member, reason="Время мута истекло")

class StatsStorage:
    def __init__(self) -> None:
        DATA_DIR.mkdir(parents=True, exist_ok=True)
        DOSSIER_DIR.mkdir(parents=True, exist_ok=True)
        self.users_file = DATA_DIR / "users.json"
        self.messages_file = DATA_DIR / "messages.jsonl"
        self.voice_file = DATA_DIR / "voice.jsonl"
        self.active_file = DATA_DIR / "active_voice.json"
        self.users: dict[str, dict] = {}
        self.active_sessions: dict[str, dict] = {}
        self._load()

    def _load(self) -> None:
        if self.users_file.exists():
            try:
                with self.users_file.open("r", encoding="utf-8") as fp:
                    self.users = json.load(fp).get("users", {})
            except: self.users = {}
        if self.active_file.exists():
            try:
                with self.active_file.open("r", encoding="utf-8") as fp:
                    self.active_sessions = json.load(fp)
            except: self.active_sessions = {}

    def _save_users(self) -> None:
        tmp = self.users_file.with_name(self.users_file.name + ".tmp")
        try:
            with tmp.open("w", encoding="utf-8") as fp:
                json.dump({"users": self.users}, fp, ensure_ascii=False, indent=2)
            tmp.replace(self.users_file)
        except: pass

    def _save_active(self) -> None:
        tmp = self.active_file.with_name(self.active_file.name + ".tmp")
        try:
            with tmp.open("w", encoding="utf-8") as fp:
                json.dump(self.active_sessions, fp, ensure_ascii=False, indent=2)
            tmp.replace(self.active_file)
        except: pass

    def update_user(self, user) -> dict:
        uid = str(user.id)
        avatar = user.display_avatar.url if getattr(user, "display_avatar", None) else ""
        first_seen = self.users.get(uid, {}).get("first_seen", time.time())
        info = {
            "id": uid, "name": str(user), "display_name": getattr(user, "display_name", str(user)),
            "avatar": avatar, "first_seen": first_seen,
            "created_at": user.created_at.timestamp() if getattr(user, "created_at", None) else None,
            "joined_at": user.joined_at.timestamp() if getattr(user, "joined_at", None) else None,
            "roles": [r.name for r in getattr(user, "roles", []) if r.name != "@everyone"],
        }
        if self.users.get(uid) != info:
            self.users[uid] = info
            self._save_users()
        return info

    def record_message(self, user) -> None:
        try:
            with self.messages_file.open("a", encoding="utf-8") as fp:
                fp.write(json.dumps({"uid": str(user.id), "ts": time.time()}, ensure_ascii=False) + "\n")
        except: pass

        uid = str(user.id)
        session = {"channel_id": str(channel.id), "channel_name": getattr(channel, "name", "?"), "start": time.time()}
        self.active_sessions[uid] = session
        self._save_active()
        return session

    def end_voice(self, user_or_id, end_ts: Optional[float] = None) -> Optional[float]:
        uid = str(user_or_id.id) if hasattr(user_or_id, "id") else str(user_or_id)
        session = self.active_sessions.pop(uid, None)
        if session is None: return None
        end_ts = end_ts or time.time()
        try:
            with self.voice_file.open("a", encoding="utf-8") as fp:
                fp.write(json.dumps({"uid": uid, "cid": session["channel_id"], "cname": session["channel_name"], "start": session["start"], "end": end_ts}, ensure_ascii=False) + "\n")
        except: pass
        self._save_active()
        return end_ts - session["start"]

    def get_stats(self, user_id) -> tuple[dict[str, int], dict[str, float]]:
        uid = str(user_id)
        now = time.time()
        periods = {"day": now - 86400, "week": now - 7 * 86400, "month": now - 30 * 86400, "year": now - 365 * 86400, "five_years": now - 5 * 365 * 86400}
        msg_counts = {k: 0 for k in periods}
        voice_seconds = {k: 0.0 for k in periods}

        if self.messages_file.exists():
            try:
                with self.messages_file.open("r", encoding="utf-8") as fp:
                    for line in fp:
                        try: d = json.loads(line)
                        except: continue
                        if d.get("uid") != uid: continue
                        for k, start in periods.items():
                            if d.get("ts", 0) >= start: msg_counts[k] += 1
            except: pass

        if self.voice_file.exists():
            try:
                with self.voice_file.open("r", encoding="utf-8") as fp:
                    for line in fp:
                        try: d = json.loads(line)
                        except: continue
                        if d.get("uid") != uid: continue
                        s, e = d.get("start", 0), d.get("end", 0)
                        for k, pstart in periods.items():
                            if max(e, pstart) > max(s, pstart): voice_seconds[k] += max(e, pstart) - max(s, pstart)
            except: pass

        active = self.active_sessions.get(uid)
        if active:
            s, e = active["start"], now
            for k, pstart in periods.items():
                if e > max(s, pstart): voice_seconds[k] += e - max(s, pstart)

        return msg_counts, voice_seconds

def enable_dave_decryption() -> None:
    """
    voice_recv не расшифровывает DAVE (E2EE). Патчим PacketRouter.feed_rtp,
    чтобы прогонять payload через встроенный метод discord.py.
    """
    try:
        from discord.ext.voice_recv import router as vr_router
    except Exception:
        log.warning("voice_recv не найден", exc_info=True)
        return

    original_feed_rtp = vr_router.PacketRouter.feed_rtp
    if getattr(original_feed_rtp, "_krim_dave", False):
        return

    warned_ssrc: set[int] = set()

    def feed_rtp(self, packet) -> None:
        try:
            state = getattr(self.reader.voice_client, "_connection", None)
            if state and getattr(state, "dave_session", None) and getattr(state, "dave_protocol_version", 0) > 0:
                user_id = self.reader.voice_client._get_id_from_ssrc(packet.ssrc)
                if user_id:
                    if hasattr(state, "_decrypt_dave_packet"):
                        # Просто вызываем метод, он меняет packet.decrypted_data "на месте"
                        state._decrypt_dave_packet(user_id, packet)
                    else:
                        # Фоллбек для старых версий discord.py
                        import davey
                        decrypted = state.dave_session.decrypt(user_id, davey.MediaType.audio, packet.decrypted_data)
                        if not decrypted:
                            return
                        packet.decrypted_data = decrypted
        except Exception as exc:
            if packet.ssrc not in warned_ssrc:
                warned_ssrc.add(packet.ssrc)
                log.debug("DAVE-дешифровка пропущена (ssrc %s): %s", packet.ssrc, exc)
            return # Пропускаем битый пакет
        
        return original_feed_rtp(self, packet)

    feed_rtp._krim_dave = True  # type: ignore[attr-defined]
    vr_router.PacketRouter.feed_rtp = feed_rtp  # type: ignore[assignment]
    log.info("DAVE-дешифровка активирована (исправленный метод discord.py)")

_instance_lock = None


def acquire_single_instance_lock() -> bool:
    """Не даёт запустить второй экземпляр: иначе бот отвечает на сообщения дважды."""
    global _instance_lock
    lock_path = BASE_DIR / "krim_logs.lock"
    try:
        handle = open(lock_path, "a+")
    except OSError as exc:
        log.warning("Не удалось открыть lock-файл %s: %s", lock_path, exc)
        return True
    try:
        if os.name == "nt":
            import msvcrt
            handle.seek(0)
            msvcrt.locking(handle.fileno(), msvcrt.LK_NBLCK, 1)
        else:
            import fcntl
            fcntl.flock(handle.fileno(), fcntl.LOCK_EX | fcntl.LOCK_NB)
    except OSError:
        handle.close()
        return False
    try:
        handle.seek(0)
        handle.truncate()
        handle.write(str(os.getpid()))
        handle.flush()
    except OSError:
        pass
    _instance_lock = handle  # держим файл открытым на всё время работы
    return True

def humans_in(channel) -> int:
    """Сколько живых (не ботов) участников в голосовом канале."""
    return sum(1 for m in getattr(channel, "members", []) if not m.bot)


def pick_voice_channel(guild):
    """Первый голосовой канал сервера, где есть люди. AFK-канал пропускаем."""
    afk_id = getattr(getattr(guild, "afk_channel", None), "id", None)
    for channel in guild.voice_channels:
        if afk_id and channel.id == afk_id:
            continue
        if humans_in(channel) > 0:
            return channel
    return None


async def disconnect_voice(vc) -> None:
    try:
        if getattr(vc, "is_listening", lambda: False)():
            vc.stop_listening()
    except Exception:
        log.debug("stop_listening завершился с ошибкой", exc_info=True)
    try:
        await vc.disconnect(force=True)
    except Exception:
        log.debug("Не удалось отключить voice-клиент", exc_info=True)


async def connect_voice(guild, channel) -> None:
    try:
        vc = await channel.connect(cls=voice_recv.VoiceRecvClient, self_deaf=True)
        vc.listen(TranscriptionSink())
        _voice_fail_at.pop(channel.id, None)
        log.info(
            "[%s] Подключился к войсу: #%s (%d чел.)",
            guild.name, channel.name, humans_in(channel),
        )
    except discord.Forbidden:
        _voice_fail_at[channel.id] = time.time()
        log.error("[%s] Нет прав на подключение к #%s", guild.name, channel.name)
    except Exception:
        _voice_fail_at[channel.id] = time.time()
        log.exception("[%s] Не удалось подключиться к #%s", guild.name, channel.name)


async def auto_voice_tick() -> None:
    """Держит бота в голосовом канале с людьми — только на разрешённых серверах."""
    if not voice_auto_enabled or not voice_guild_ids:
        return

    for guild in client.guilds:
        if guild.id not in voice_guild_ids:
            continue

        vc = guild.voice_client
        # Уже сидим в канале, где есть люди — ничего не делаем.
        if vc and vc.channel and humans_in(vc.channel) > 0:
            continue

        target = pick_voice_channel(guild)

        if target is None:
            if vc and vc.channel:
                log.info("[%s] В войсе никого не осталось — отключаюсь", guild.name)
                await disconnect_voice(vc)
            continue

        if vc and vc.channel and vc.channel.id == target.id:
            continue

        if time.time() - _voice_fail_at.get(target.id, 0) < VOICE_CONNECT_RETRY:
            continue

        if vc:
            await disconnect_voice(vc)
        await connect_voice(guild, target)


async def auto_voice_worker() -> None:
    await client.wait_until_ready()
    while not client.is_closed():
        try:
            await auto_voice_tick()
        except asyncio.CancelledError:
            raise
        except Exception:
            log.exception("Ошибка авто-транскрипции")
        await asyncio.sleep(VOICE_CHECK_INTERVAL)

POLITENESS_TAG = re.compile(r"\[POLITENESS:\s*([+-]?\d+)\s*\]", re.IGNORECASE)
BLOCK_TAG = re.compile(r"\[BLOCK\]", re.IGNORECASE)
DOSSIER_TAG = re.compile(r"\[DOSSIER:\s*([^\]]+)\]", re.IGNORECASE)
CLEAR_TAG = re.compile(r"\[CLEAR:\s*(\d+)\]", re.IGNORECASE)
MUTE_TAG = re.compile(r"\[MUTE:\s*(\d+):\s*([a-zA-Z0-9]+)\]", re.IGNORECASE)
UNMUTE_TAG = re.compile(r"\[UNMUTE:\s*(\d+)\]", re.IGNORECASE)
ANY_TAG = re.compile(r"\[(?:POLITENESS:\s*[+-]?\d+|BLOCK|DOSSIER:\s*[^\]]+|CLEAR:\s*\d+|MUTE:\s*\d+:\s*[a-zA-Z0-9]+|UNMUTE:\s*\d+)\]", re.IGNORECASE)

def parse_ai_tags(text: str) -> dict:
    result = {"clean": text, "politeness_delta": 0, "block": False, "dossier_targets": [], "clear": None, "mute": None, "unmute": None}
    for m in POLITENESS_TAG.findall(text):
        try: result["politeness_delta"] += int(m)
        except: pass
    if BLOCK_TAG.search(text): result["block"] = True
    result["dossier_targets"] = [t.strip() for t in DOSSIER_TAG.findall(text) if t.strip()][:3]
    if CLEAR_TAG.search(text): result["clear"] = int(CLEAR_TAG.search(text).group(1))
    if MUTE_TAG.search(text): result["mute"] = {"user_id": int(MUTE_TAG.search(text).group(1)), "time": MUTE_TAG.search(text).group(2)}
    if UNMUTE_TAG.search(text): result["unmute"] = int(UNMUTE_TAG.search(text).group(1))
    result["clean"] = ANY_TAG.sub("", text).strip()
    return result

async def call_ai(
    messages: list[dict],
    *,
    model: Optional[str] = None,
    max_tokens: Optional[int] = None,
    temperature: Optional[float] = None,
    timeout: int = 90,
) -> Optional[str]:
    """Запрос к OpenAI-совместимому API. Параметры можно переопределить для DeepNeuro."""
    if not AI_API_KEY: return None
    url = f"{ai_config.get('base_url', 'https://api.relaymodels.com/v1').rstrip('/')}/chat/completions"
    headers = {"Authorization": f"Bearer {AI_API_KEY}", "Content-Type": "application/json"}
    payload = {
        "model": model or ai_config.get("model", "gpt-4o-mini"),
        "messages": messages,
        "max_tokens": max_tokens if max_tokens is not None else ai_config.get("max_tokens", 1200),
        "temperature": temperature if temperature is not None else ai_config.get("temperature", 0.7),
    }
    try:
        async with aiohttp.ClientSession() as session:
            async with session.post(
                url,
                headers=headers,
                json=payload,
                timeout=aiohttp.ClientTimeout(total=max(30, int(timeout))),
            ) as resp:
                if resp.status != 200:
                    log.warning("AI API вернул HTTP %s", resp.status)
                    return None
                data = await resp.json()
                return data["choices"][0]["message"]["content"]
    except Exception:
        log.exception("Ошибка запроса к AI API")
        return None

async def resolve_user_target(target: str, guild) -> Optional[int]:
    target = target.strip()
    m = re.match(r"^<@!?(\d{17,20})>$", target)
    if m: return int(m.group(1))
    if target.isdigit() and 17 <= len(target) <= 20: return int(target)
    if not guild: return None
    clean = target.lstrip("@").strip().lower()
    for member in guild.members:
        if clean in {member.name.lower(), (member.nick or "").lower(), member.display_name.lower(), str(member).lower()}: return member.id
    return None

_DOCX_NS = "{http://schemas.openxmlformats.org/wordprocessingml/2006/main}"


def _decode_bytes(raw: bytes) -> str:
    for enc in ("utf-8-sig", "utf-8", "cp1251", "koi8-r", "latin-1"):
        try:
            return raw.decode(enc)
        except UnicodeDecodeError:
            continue
    return raw.decode("utf-8", errors="replace")


def _docx_to_text(raw: bytes) -> str:
    """Достаёт текст из .docx (это zip с word/document.xml) без внешних библиотек."""
    try:
        import zipfile
        import xml.etree.ElementTree as ET

        with zipfile.ZipFile(io.BytesIO(raw)) as zf:
            parts: list[str] = []
            for name in ("word/document.xml", "word/footnotes.xml", "word/endnotes.xml"):
                if name not in zf.namelist():
                    continue
                root = ET.fromstring(zf.read(name))
                for paragraph in root.iter(_DOCX_NS + "p"):
                    text = "".join(
                        node.text or "" for node in paragraph.iter(_DOCX_NS + "t")
                    )
                    if text.strip():
                        parts.append(text)
            return "\n".join(parts)
    except Exception as exc:
        log.warning("Не удалось прочитать .docx: %s", exc)
        return ""


async def extract_attachment_text(attachment) -> Optional[str]:
    """Возвращает текстовый блок для одного вложения или None, если формат не поддержан."""
    filename = attachment.filename or "file"
    ext = Path(filename).suffix.lower()
    content_type = attachment.content_type or ""
    is_docx = ext in DOCX_EXTENSIONS
    is_text = ext in TEXT_EXTENSIONS or content_type.startswith("text/") or content_type in (
        "application/json", "application/xml", "application/x-yaml", "application/javascript",
    )
    if not (is_docx or is_text):
        return None

    size = getattr(attachment, "size", 0) or 0
    if size > ATTACH_MAX_BYTES:
        return f"--- Файл: {filename} ---\n[пропущен: слишком большой, {size} байт]"

    try:
        raw = await attachment.read()
    except Exception as exc:
        log.warning("Не удалось скачать вложение %s: %s", filename, exc)
        return f"--- Файл: {filename} ---\n[не удалось скачать: {exc}]"

    if not is_docx and b"\x00" in raw[:2048]:
        return f"--- Файл: {filename} ---\n[пропущен: похоже на бинарный файл]"

    text = _docx_to_text(raw) if is_docx else _decode_bytes(raw)
    if not text.strip():
        return None
    if len(text) > ATTACH_ONE_FILE_CHARS:
        text = text[:ATTACH_ONE_FILE_CHARS] + "\n…[файл обрезан]"
    return f"--- Файл: {filename} ---\n{text}"


async def extract_attachments_text(message: discord.Message) -> str:
    """Собирает текст всех поддерживаемых вложений сообщения в один блок."""
    attachments = getattr(message, "attachments", None) or []
    if not attachments:
        return ""

    blocks: list[str] = []
    total = 0
    skipped = 0
    for attachment in attachments[:ATTACH_MAX_FILES]:
        block = await extract_attachment_text(attachment)
        if not block:
            skipped += 1
            continue
        if total + len(block) > ATTACH_TOTAL_CHARS:
            remaining = ATTACH_TOTAL_CHARS - total
            if remaining > 300:
                blocks.append(block[:remaining] + "\n…[обрезано]")
            skipped += 1
            break
        blocks.append(block)
        total += len(block)

    if not blocks:
        return ""
    header = "[ПРИКРЕПЛЁННЫЕ ФАЙЛЫ — используй их как контекст для ответа]"
    if skipped:
        header += f" (пропущено файлов: {skipped})"
    return header + "\n\n" + "\n\n".join(blocks)


async def handle_ai(message: discord.Message, is_bot_chat: bool = False) -> None:
    if not AI_API_KEY or not client.user: return
    content = re.sub(rf"<@!?{client.user.id}>", "", message.content).strip()
    content = _resolve_mentions(content, message.guild).strip()
    attachments_text = await extract_attachments_text(message)
    if attachments_text:
        content = f"{content}\n\n{attachments_text}".strip() if content else attachments_text
    if not content: return

    user_id = message.author.id
    state = await get_user_state(user_id)
    just_unblocked = False

    if not is_bot_chat and state["is_blocked"]:
        if detect_apology(content):
            await set_user_blocked(user_id, False)
            await set_politeness(user_id, ai_config.get("politeness", {}).get("restore_on_apology", 100))
            just_unblocked = True
        else:
            await _notify_blocked(message)
            return

    await save_ai_message(user_id, client.user.id, message.channel.id, "user", content)
    history = await get_context(user_id, client.user.id, message.channel.id, ai_config.get("context_limit", 20))
    politeness_now = (await get_user_state(user_id))["politeness"]
    messages = [{"role": "system", "content": build_system_prompt(message.author, politeness_now)}]

    elif just_unblocked: messages.append({"role": "system", "content": f"[СИСТЕМА] Блокировка снята. Поприветствуй возвращение."})
    messages.extend(history)

    raw_reply = await call_ai(messages)
    if not raw_reply:
        if not is_bot_chat: await message.reply("Не удалось получить ответ от ИИ.", mention_author=False)
        return

    parsed = parse_ai_tags(raw_reply)
    if parsed["dossier_targets"]:
        dossier_blocks = []
        for target in parsed["dossier_targets"]:
            target_id = await resolve_user_target(target, message.guild)
            if target_id: dossier_blocks.append(await get_dossier_text(target_id))

            messages.append({"role": "assistant", "content": raw_reply})
            messages.append({"role": "system", "content": "[ДОСЬЕ]\n\n" + "\n\n---\n\n".join(dossier_blocks)})

                parsed = parse_ai_tags(second)

    if is_admin(user_id) and message.guild:

            try: await message.channel.purge(limit=min(parsed["clear"], 100))
            except discord.Forbidden: await message.channel.send("⚠️ Не удалось очистить чат.")
        if parsed["mute"]:
            secs = parse_duration(parsed["mute"]["time"])
            if secs > 0 and (member := message.guild.get_member(parsed["mute"]["user_id"])):
                success, err = await apply_mute(message.guild, member, secs, reason=f"AI mute by {message.author}")
                if not success: await message.channel.send(f"⚠️ Не удалось замутить: {err}")
        if parsed["unmute"]:

def split_message(text, limit=MESSAGE_LIMIT):
    if len(text) <= limit: return [text]
    parts, current = [], ""
    for line in text.splitlines(keepends=True):
        while len(line) > limit:
            if current: parts.append(current); current = ""
            parts.append(line[:limit]); line = line[limit:]
        if len(current) + len(line) > limit: parts.append(current); current = line
        else: current += line
    if current: parts.append(current)
    return [p for p in parts if p.strip()]

def format_duration(seconds):
    s = int(max(0, seconds))
    if s < 60: return f"{s} сек"
    m, s2 = divmod(s, 60)
    if m < 60: return f"{m} мин {s2} сек" if s2 else f"{m} мин"
    h, m2 = divmod(m, 60)
    if h < 24: return f"{h} ч {m2} мин" if m2 else f"{h} ч"
    d, h2 = divmod(h, 24)
    return f"{d} д {h2} ч" if h2 else f"{d} д"

_geo_cache: dict[str, str] = {}


async def resolve_host_ip(host: str) -> str:
    """Возвращает IP для домена/адреса; при ошибке — исходную строку."""
    host = (host or "").strip().strip("[]")
    if not host:
        return "?"
    try:
        return await asyncio.to_thread(socket.gethostbyname, host)
    except OSError:
        return host


async def geo_country(ip: str) -> str:
    """Страна по IP через бесплатный ipwho.is. Результат кэшируется."""
    if not ip or ip == "?":
        return "неизвестно"
    if ip in _geo_cache:
        return _geo_cache[ip]
    try:
        async with aiohttp.ClientSession() as session:
            async with session.get(
                f"https://ipwho.is/{ip}",
                timeout=aiohttp.ClientTimeout(total=8),
            ) as response:
                if response.status != 200:
                    return "неизвестно"
                data = await response.json()
        if not data.get("success", False):
            return "неизвестно"
        country = data.get("country") or "неизвестно"
        code = data.get("country_code") or ""
        result = f"{country} ({code})" if code else country
        _geo_cache[ip] = result
        return result
    except Exception:
        log.debug("Не удалось определить страну по IP %s", ip, exc_info=True)
        return "неизвестно"


async def tcp_ping(
    host: str, port: int = 443, attempts: int = 3, timeout: float = 5.0
) -> tuple[str, Optional[float]]:
    """Минимальная задержка TCP-подключения из нескольких попыток."""
    ip = await resolve_host_ip(host)
    target = ip if ip and ip != "?" else host
    best: Optional[float] = None
    for _ in range(attempts):
        writer = None
        started = time.perf_counter()
        try:
            _reader, writer = await asyncio.wait_for(
                asyncio.open_connection(target, port), timeout=timeout
            )
            elapsed = (time.perf_counter() - started) * 1000.0
            if best is None or elapsed < best:
                best = elapsed
        except Exception:
            pass
        finally:
            if writer is not None:
                try:
                    writer.close()
                    await writer.wait_closed()
                except Exception:
                    pass
    return ip, best

_DEEPN_ARGS_RE = re.compile(
    r"^(?:\"((?:\\.|[^\"\\])*)\"|'((?:\\.|[^'\\])*)')\s+(\d+)\s*$",
    re.DOTALL,
)


def parse_deepneuro_args(raw: str) -> Optional[tuple[str, int]]:
    """Разбирает '«промпт» 3' -> ('промпт', 3). Промпт обязан быть в кавычках."""
    raw = (raw or "").strip()
    if not raw:
        return None
    match = _DEEPN_ARGS_RE.match(raw)
    if not match:
        return None
    prompt = match.group(1) if match.group(1) is not None else match.group(2)
    prompt = prompt.replace('\\"', '"').replace("\\'", "'").strip()
    steps = int(match.group(3))
    if not prompt or not 1 <= steps <= 4:
        return None
    return prompt, steps

            self.uuid,
            self.address,
            self.port,
            self.network,
            self.security,
            self.path,
            self.sni,
            self.service_name,
        )

    @property
    def name(self) -> str:
        if self.label:
            return self.label
        return f"{self.address}:{self.port} [{self.network}/{self.security}]"

    def score(self) -> int:
        """Чем больше, тем раньше пробуем конфиг."""
        score = 0
        if self.security == "reality":
            score += 40
        elif self.security == "tls":
            score += 25
        if self.network in ("tcp", "raw"):
            score += 15
        elif self.network == "ws":
            score += 12
        elif self.network == "grpc":
            score -= 5
        elif self.network == "httpupgrade":
            score -= 8
        elif self.network == "xhttp":
            score -= 15
        if self.flow:
            score += 5
        return score

    def outbound(self) -> dict:
        network = "tcp" if self.network in ("tcp", "raw") else self.network
        stream: dict[str, Any] = {"network": network, "security": self.security}

        if self.security == "tls":
            tls: dict[str, Any] = {
                "serverName": self.sni or self.host or self.address,
                "allowInsecure": False,

                "publicKey": self.public_key,
                "shortId": self.short_id,
                "spiderX": "/",
                "show": False,

            stream["tcpSettings"] = {"header": {"type": self.header_type or "none"}}
        elif network == "ws":
            stream["wsSettings"] = {
                "path": self.path or "/",
                "headers": {"Host": self.host or self.sni or self.address},
            }
        elif network == "grpc":
            stream["grpcSettings"] = {
                "serviceName": self.service_name,
                "multiMode": self.mode == "multi",
            }
        elif network == "httpupgrade":
            stream["httpupgradeSettings"] = {
                "path": self.path or "/",
                "host": self.host or self.sni or self.address,
            }
        elif network == "xhttp":

                "path": self.path or "/",

                "mode": self.mode or "auto",

                "path": self.path or "/",
                "host": [self.host or self.sni or self.address],
            }

        user: dict[str, Any] = {"id": self.uuid, "encryption": "none"}
        if self.flow:
            user["flow"] = self.flow
        if self.packet_encoding:
            user["packetEncoding"] = self.packet_encoding

        return {
            "tag": "proxy",
            "protocol": "vless",
            "settings": {
                "vnext": [
                    {
                        "address": self.address,
                        "port": self.port,
                        "users": [user],
                    }
                ]
            },
            "streamSettings": stream,
        }


def parse_vless(uri: str) -> Optional[VlessConfig]:
    """Разбирает vless:// ссылку. Возвращает None, если конфиг не поддержан."""
    uri = (uri or "").strip()
    if not uri.lower().startswith("vless://"):
        return None
    try:
        parts = urlsplit(uri)
    except ValueError:
        return None

    if parts.scheme.lower() != "vless" or not parts.username or not parts.hostname:
        return None
    try:
        port = int(parts.port or 443)
    except (TypeError, ValueError):
        return None

    query = {k.lower(): v for k, v in parse_qs(parts.query, keep_blank_values=True).items()}

    def q1(*names: str, default: str = "") -> str:
        for name in names:
            values = query.get(name.lower())
            if values:
                return values[0]
        return default

    network = q1("type", "network", default="tcp").lower()
    if network not in SUPPORTED_NETWORKS:
        return None

    security = q1("security", default="none").lower()
    if security not in ("none", "tls", "reality"):
        return None

    public_key = q1("pbk")
    if security == "reality" and not public_key:
        return None

    extra: Optional[dict] = None
    raw_extra = q1("extra")
    if raw_extra:
        try:
            parsed_extra = json.loads(raw_extra)
            if isinstance(parsed_extra, dict):
                extra = parsed_extra
        except (TypeError, ValueError):
            extra = None

    return VlessConfig(
        raw=uri,
        uuid=parts.username,
        address=parts.hostname,
        port=port,
        network=network,
        security=security,
        sni=q1("sni"),
        host=q1("host"),
        path=q1("path", default="/") or "/",
        service_name=q1("servicename"),
        flow=q1("flow"),
        fingerprint=q1("fp", default="chrome") or "chrome",
        public_key=public_key,
        short_id=q1("sid"),
        mode=q1("mode"),
        header_type=q1("headertype", default="none") or "none",
        alpn=q1("alpn"),
        packet_encoding=q1("packetencoding"),
        extra=extra,
        label=unquote(parts.fragment or ""),
    )

class NetBypassManager:
    """Единая точка управления VLESS-туннелем и пулом HTTPS-прокси."""

    def __init__(self, config: dict, base_dir: Path, logger: logging.Logger):
        self.config: dict = config or {}
        self.base_dir = Path(base_dir)
        self.log = logger.getChild("netbypass") if hasattr(logger, "getChild") else logger

        self._mode: Optional[str] = None
        self._proxy_url: Optional[str] = None
        self._client: Any = None

        self._stop = asyncio.Event()
        self._watchdog_task: Optional[asyncio.Task] = None
        self._refresh_lock = asyncio.Lock()
        self._rotate_lock = asyncio.Lock()

        self._xray_proc: Optional[asyncio.subprocess.Process] = None
        self._xray_log: Optional[Any] = None
        self._xray_port: Optional[int] = None
        self._xray_cfg_path: Optional[Path] = None
        self._xray_exe: Optional[str] = None

        self._vless_pool: list[VlessConfig] = []
        self._vless_index: int = -1
        self._backups: list[str] = []
        self._last_refresh: float = 0.0

        data_dir = self.base_dir / "data" / "netbypass"
        self._cache_file = data_dir / "subscription_cache.txt"
        self._last_good_file = data_dir / "last_good.txt"
        self._proxy_cache_file = data_dir / "proxy_pool_cache.json"

    # -- публичный API ------------------------------------------------------

    @property
    def proxy_url(self) -> Optional[str]:
        return self._proxy_url

    @property
    def enabled(self) -> bool:
        return bool(self.config.get("enabled", False))

    def describe(self) -> str:
        if not self._proxy_url:
            return "выключен"
        if self._mode == "vless":
            cfg = None
            if 0 <= self._vless_index < len(self._vless_pool):
                cfg = self._vless_pool[self._vless_index]
            return f"VLESS ({cfg.name if cfg else '?'}) через {self._proxy_url}"
        return (
            f"HTTPS-прокси {self._proxy_url} "
            f"(запасных: {len(self._backups)}/{self._pool_size})"
        )

    def status(self) -> dict:
        """Машиночитаемое состояние для команды log!proxy."""
        vless = None
        if 0 <= self._vless_index < len(self._vless_pool):
            cfg = self._vless_pool[self._vless_index]
            vless = {
                "name": cfg.name,
                "address": cfg.address,
                "port": cfg.port,
                "network": cfg.network,
                "security": cfg.security,
            }
        return {
            "enabled": self.enabled,
            "mode": self._mode,
            "proxy_url": self._proxy_url,
            "vless": vless,
            "backups": list(self._backups),
            "pool_size": self._pool_size,
        }

    def attach_client(self, client: Any) -> None:
        """Привязывает discord.py-клиент, чтобы менеджер мог менять прокси."""
        self._client = client
        try:
            client.http.proxy = self._proxy_url
        except Exception:
            self.log.debug("Не удалось сразу выставить прокси в клиенте", exc_info=True)

    async def start(self) -> Optional[str]:
        """Поднимает выбранный режим. Возвращает URL HTTP-прокси или None."""
        if not self.enabled:
            self.log.info("Сетевой обход отключён (network.enabled=false)")
            return None

        self._stop.clear()
        mode = str(self.config.get("mode", "vless")).strip().lower()
        self.log.info("Сетевой обход включён, режим: %s", mode)

        url: Optional[str] = None
        try:
            if mode == "vless":
                url = await self._start_vless()
            elif mode == "proxy":
                url = await self._start_proxy()
            else:
                self.log.error(
                    "Неизвестный network.mode=%r (ожидается 'vless' или 'proxy')", mode
                )
        except Exception:
            self.log.exception("Ошибка запуска режима %s", mode)

        fallback = str(self.config.get("fallback", "proxy")).strip().lower()
        if not url and fallback in ("proxy", "vless") and fallback != mode:
            self.log.warning(
                "Режим %s не поднялся — пробую резервный режим %s", mode, fallback
            )
            try:
                url = (
                    await self._start_proxy()
                    if fallback == "proxy"
                    else await self._start_vless()
                )
            except Exception:
                self.log.exception("Резервный режим %s не поднялся", fallback)

        if not url:
            self.log.error(
                "Не удалось поднять сетевой обход — работаю напрямую. "
                "Discord может быть недоступен без системного VPN."
            )
            return None

        self._apply(url, "start")
        self._watchdog_task = asyncio.create_task(
            self._watchdog(), name="netbypass-watchdog"
        )
        self.log.info("Сетевой обход активен: %s", self.describe())
        return url

    async def stop(self) -> None:
        self._stop.set()
        task = self._watchdog_task
        self._watchdog_task = None
        if task is not None:
            task.cancel()
            try:
                await task
            except (asyncio.CancelledError, Exception):
                pass
        await self._stop_xray()
        if self._client is not None:
            try:
                self._client.http.proxy = None
            except Exception:
                pass

        except (TypeError, ValueError):

    @property
    def _test_timeout(self) -> int:
        return max(2, self._int(self._section("proxy"), "test_timeout", 8))

    @property
    def _pool_size(self) -> int:
        return max(1, self._int(self._section("proxy"), "pool_size", 4))

    @property
    def _concurrency(self) -> int:
        return max(1, self._int(self._section("proxy"), "concurrency", 60))

    @property

        return max(1, self._int(self._section("proxy"), "max_candidates", 300))

    @property

        return max(5, self._int(self._section("proxy"), "cache_limit", 50))

    @property

    @property

        return max(3, self._int(self._section("vless"), "startup_timeout", 8))

        value = self._int(self._section("vless"), "local_port", 0)
        return value if 0 < value < 65536 else 0

    @property
    def _verify_attempts(self) -> int:
        return max(1, min(3, self._int(self._section("proxy"), "verify_attempts", 2)))

    # -- проверка доступности ----------------------------------------------

    async def _test_proxy(
        self,
        proxy_url: str,
        session: Optional[aiohttp.ClientSession] = None,
        confirm: Optional[int] = None,
    ) -> Optional[float]:
        """Проверяет прокси запросом к Discord по HTTPS. Возвращает мс или None.

        confirm — сколько последовательных запросов должен выдержать прокси.
        Для пула бесплатных прокси двух проверок достаточно, чтобы отсеять
        сервера, которые отдают ровно один ответ и сразу умирают.
        """
        attempts = self._verify_attempts if confirm is None else max(1, confirm)
        own_session = session is None
        if own_session:
            session = aiohttp.ClientSession(
                timeout=aiohttp.ClientTimeout(total=self._test_timeout),
                connector=aiohttp.TCPConnector(force_close=True),
            )
        try:
            latency: Optional[float] = None
            for _ in range(attempts):
                current = await self._single_proxy_request(proxy_url, session)
                if current is None:
                    return None
                if latency is None:
                    latency = current
            return latency
        finally:
            if own_session:
                await session.close()

    async def _single_proxy_request(
        self, proxy_url: str, session: aiohttp.ClientSession
    ) -> Optional[float]:
        try:
            started = time.perf_counter()
            timeout = aiohttp.ClientTimeout(total=self._test_timeout)
            async with session.get(
                self._test_url,
                proxy=proxy_url,
                timeout=timeout,
                allow_redirects=False,
                headers={"Connection": "close"},
            ) as response:
                if response.status not in (200, 204):
                    return None
                body = await response.read()
            # Discord gateway отвечает JSON вида {"url": "wss://gateway.discord.gg"}
            if b'"url"' not in body and b"discord" not in body.lower():
                return None
            return (time.perf_counter() - started) * 1000.0
        except (aiohttp.ClientError, asyncio.TimeoutError, OSError, ValueError):
            return None
        except Exception:
            self.log.debug("Ошибка проверки прокси %s", proxy_url, exc_info=True)
            return None

    # -- режим VLESS --------------------------------------------------------

    async def _start_vless(self) -> Optional[str]:
        self._mode = "vless"
        pool = await self._load_vless_configs()
        if not pool:
            self.log.error(
                "VLESS: не удалось получить конфиги из подписки "
                "(заполните network.vless.subscription в config.json)"
            )
            return None

        exe = await asyncio.to_thread(self._ensure_xray)
        if not exe:
            self.log.error(
                "VLESS: ядро Xray-core не найдено. Включите network.vless.auto_download "
                "или укажите network.vless.core_path"
            )
            return None

        self._xray_exe = exe
        self._vless_pool = pool
        self._vless_index = -1

        last_good = self._read_last_good()
        if last_good:
            for position, cfg in enumerate(pool):
                if cfg.raw == last_good:
                    pool.insert(0, pool.pop(position))
                    self.log.info("VLESS: последний рабочий конфиг поднят в начало очереди")
                    break

        self.log.info(
            "VLESS: загружено %d уникальных конфигов, проверяю пачками по %d",
            len(pool),
            self._probe_concurrency,
        )
        return await self._try_vless_from(0, wrap=False)

    async def _try_vless_from(self, start: int, wrap: bool) -> Optional[str]:
        async with self._rotate_lock:
            total = len(self._vless_pool)
            if total == 0:
                return None

            # Старый (уже нерабочий) туннель больше не нужен.
            await self._stop_xray()

            max_attempts = self._int(self._section("vless"), "max_attempts", 0)
            limit = total if max_attempts <= 0 else min(max_attempts, total)
            order = [(start + offset) % total for offset in range(limit)]
            batch_size = 1 if self._local_port else self._probe_concurrency

            for offset in range(0, len(order), batch_size):
                batch = order[offset : offset + batch_size]
                self.log.info(
                    "VLESS: проверяю конфиги %d-%d из %d",
                    offset + 1,
                    offset + len(batch),
                    len(order),
                )
                winner = await self._probe_batch(batch)
                if winner is None:
                    continue

                self._xray_proc = winner.proc
                self._xray_log = winner.log_handle
                self._xray_port = winner.port
                self._xray_cfg_path = winner.config_path
                self._vless_index = winner.index
                self._save_last_good(winner.cfg)
                self.log.info(
                    "VLESS: подключено через %s (%.0f мс, порт %d)",
                    winner.cfg.name,
                    winner.latency,
                    winner.port,
                )
                return winner.proxy_url

            self.log.warning("VLESS: проверено %d конфигов, рабочих нет", len(order))
            return None

    async def _probe_batch(self, indices: list[int]) -> Optional[_ProbeResult]:
        ports = self._pick_ports(len(indices))
        tasks = [
            self._probe_config(index, self._vless_pool[index], port)
            for index, port in zip(indices, ports)
        ]
        results = await asyncio.gather(*tasks, return_exceptions=True)
        successful = [item for item in results if isinstance(item, _ProbeResult)]
        if not successful:
            return None
        successful.sort(key=lambda item: item.latency)
        winner = successful[0]
        for extra in successful[1:]:
            await self._terminate_proc(extra.proc, extra.log_handle)
        return winner

    async def _probe_config(
        self, index: int, cfg: VlessConfig, port: int
    ) -> Optional[_ProbeResult]:
        data_dir = self.base_dir / "data" / "netbypass"
        try:
            data_dir.mkdir(parents=True, exist_ok=True)
        except OSError as exc:
            self.log.error("VLESS: не удалось создать каталог %s: %s", data_dir, exc)
            return None

        config_path = data_dir / f"xray-{port}.json"
        xray_config = {
            "log": {"loglevel": "warning"},
            "inbounds": [
                {
                    "listen": "127.0.0.1",
                    "port": port,
                    "protocol": "http",
                    "settings": {"timeout": 300},
                }
            ],
            "outbounds": [cfg.outbound()],
        }
        try:
            config_path.write_text(
                json.dumps(xray_config, ensure_ascii=False, indent=2),
                encoding="utf-8",
            )
        except OSError as exc:
            self.log.error("VLESS: не удалось записать конфиг Xray: %s", exc)
            return None

        log_path = data_dir / f"xray-{port}.log"
        try:
            log_handle = open(log_path, "ab")
        except OSError as exc:
            self.log.error("VLESS: не удалось открыть лог Xray: %s", exc)
            return None

        proc: Optional[asyncio.subprocess.Process] = None
        try:
            kwargs: dict[str, Any] = {}
            if os.name == "nt" and hasattr(subprocess, "CREATE_NO_WINDOW"):
                kwargs["creationflags"] = subprocess.CREATE_NO_WINDOW
            assert self._xray_exe is not None
            proc = await asyncio.create_subprocess_exec(
                self._xray_exe,
                "run",
                "-c",
                str(config_path),
                stdout=log_handle,
                stderr=log_handle,
                **kwargs,
            )

            if not await self._wait_process_port(proc, port, self._startup_timeout):
                self.log.debug(
                    "VLESS: Xray не открыл порт %d (код %s), лог: %s",
                    port,
                    proc.returncode,
                    log_path,
                )
                await self._terminate_proc(proc, log_handle)
                return None

            proxy_url = f"http://127.0.0.1:{port}"
            latency = await self._test_proxy(proxy_url, confirm=1)
            if latency is None:
                await self._terminate_proc(proc, log_handle)
                return None

            return _ProbeResult(
                index=index,
                cfg=cfg,
                latency=latency,
                proxy_url=proxy_url,
                port=port,
                proc=proc,
                log_handle=log_handle,
                config_path=config_path,
            )
        except Exception:
            self.log.debug("VLESS: ошибка проверки конфига %s", cfg.name, exc_info=True)
            if proc is not None:
                await self._terminate_proc(proc, log_handle)
            else:
                try:
                    log_handle.close()
                except Exception:
                    pass
            return None

    async def _stop_xray(self) -> None:
        proc = self._xray_proc
        log_handle = self._xray_log
        self._xray_proc = None
        self._xray_log = None
        self._xray_port = None
        self._xray_cfg_path = None
        await self._terminate_proc(proc, log_handle)

    async def _terminate_proc(
        self,
        proc: Optional[asyncio.subprocess.Process],
        log_handle: Optional[Any],
    ) -> None:
        if proc is not None and proc.returncode is None:
            try:
                proc.terminate()
            except ProcessLookupError:
                pass
            try:
                await asyncio.wait_for(proc.wait(), timeout=5)
            except (asyncio.TimeoutError, ProcessLookupError):
                try:
                    proc.kill()
                except ProcessLookupError:
                    pass
                try:
                    await proc.wait()
                except Exception:
                    pass
        if log_handle is not None:
            try:
                log_handle.close()
            except Exception:
                pass

    async def _wait_process_port(
        self, proc: asyncio.subprocess.Process, port: int, timeout: int
    ) -> bool:
        loop = asyncio.get_running_loop()
        deadline = loop.time() + timeout
        while loop.time() < deadline:
            if proc.returncode is not None:
                return False
            try:
                _reader, writer = await asyncio.open_connection("127.0.0.1", port)
                writer.close()
                try:
                    await writer.wait_closed()
                except Exception:
                    pass
                return True
            except OSError:
                await asyncio.sleep(0.25)
        return False

    def _pick_port(self) -> int:
        configured = self._local_port
        if configured:
            return configured
        with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as sock:
            sock.bind(("127.0.0.1", 0))
            return int(sock.getsockname()[1])

    def _pick_ports(self, count: int) -> list[int]:
        if self._local_port:
            # Фиксированный порт нельзя занять несколькими процессами сразу,
            # поэтому при local_port проверка всегда последовательная.
            return [self._local_port]
        ports: list[int] = []
        while len(ports) < count:
            port = self._pick_port()
            if port not in ports:
                ports.append(port)
        return ports

    async def _vless_health(self) -> None:
        if not self._proxy_url:
            return
        if await self._test_proxy(self._proxy_url) is not None:
            return

        self.log.warning("VLESS: туннель перестал отвечать — ротация конфига")
        url = await self._try_vless_from(self._vless_index + 1, wrap=True)
        if url:
            self._apply(url, "rotate")
            return

        self.log.warning("VLESS: все конфиги из текущей подписки не работают, обновляю её")
        pool = await self._load_vless_configs(force_refresh=True)
        if pool:
            self._vless_pool = pool
            url = await self._try_vless_from(0, wrap=True)
            if url:
                self._apply(url, "rescan")
                return

        self.log.error("VLESS: рабочих конфигов нет — перехожу на пул HTTPS-прокси")
        self._mode = "proxy"
        new_url = await self._refresh_proxy_pool(keep_active=False)
        if new_url:
            self._apply(new_url, "fallback")

    async def _load_vless_configs(self, force_refresh: bool = False) -> list[VlessConfig]:
        text = await self._load_subscription_text(force_refresh=force_refresh)
        configs: list[VlessConfig] = []
        seen: set[tuple] = set()
        for line in text.splitlines():
            line = line.strip()
            if not line.lower().startswith("vless://"):
                continue
            try:
                cfg = parse_vless(line)
            except Exception:
                self.log.debug("Не удалось разобрать VLESS-ссылку", exc_info=True)
                continue
            if cfg is None or cfg.key in seen:
                continue
            seen.add(cfg.key)
            configs.append(cfg)
        configs.sort(key=lambda item: item.score(), reverse=True)
        return configs

    def _save_last_good(self, cfg: VlessConfig) -> None:
        try:
            self._last_good_file.parent.mkdir(parents=True, exist_ok=True)
            self._last_good_file.write_text(cfg.raw, encoding="utf-8")
        except OSError:
            self.log.debug(
                "Не удалось сохранить последний рабочий VLESS-конфиг", exc_info=True
            )

    def _read_last_good(self) -> str:
        try:
            if self._last_good_file.exists():
                return self._last_good_file.read_text(
                    encoding="utf-8", errors="ignore"
                ).strip()
        except OSError:
            pass
        return ""

    async def _load_subscription_text(self, force_refresh: bool = False) -> str:
        """Собирает VLESS-конфиги из config.json.

        network.vless.subscription может быть:
          * строкой с URL подписки;
          * строкой с одной или несколькими vless:// ссылками;
          * списком строк (URL и/или vless://), что удобно для вставки.
        Старые поля subscription_file/subscription_url поддерживаются только
        для обратной совместимости.
        """
        vless_cfg = self._section("vless")
        raw_subscription = vless_cfg.get("subscription")

        subscription_items: list[str] = []
        if isinstance(raw_subscription, str):
            subscription_items = [raw_subscription]
        elif isinstance(raw_subscription, (list, tuple)):
            subscription_items = [str(item) for item in raw_subscription]

        raw_lines: list[str] = []
        subscription_urls: list[str] = []
        for item in subscription_items:
            for line in item.splitlines():
                line = line.strip()
                if not line:
                    continue
                if line.startswith("vless://"):
                    raw_lines.append(line)
                elif line.startswith(("http://", "https://")):
                    subscription_urls.append(line)

        # Старые поля читаем только если новое поле subscription не заполнено.
        if not subscription_items:
            legacy_file = str(vless_cfg.get("subscription_file") or "").strip()
            if legacy_file:
                path = Path(legacy_file)
                if not path.is_absolute():
                    path = self.base_dir / path
                try:
                    if path.exists():
                        legacy_text = path.read_text(
                            encoding="utf-8", errors="ignore"
                        )
                        for line in legacy_text.splitlines():
                            line = line.strip()
                            if line.startswith("vless://"):
                                raw_lines.append(line)
                            elif line.startswith(("http://", "https://")):
                                subscription_urls.append(line)
                except OSError as exc:
                    self.log.warning("VLESS: не удалось прочитать %s: %s", path, exc)

            legacy_url = str(vless_cfg.get("subscription_url") or "").strip()
            if legacy_url:
                subscription_urls.append(legacy_url)

        if not subscription_urls and not raw_lines:
            subscription_urls.append(DEFAULT_SUBSCRIPTION_URL)

        fetched_parts: list[str] = []
        for url in dict.fromkeys(subscription_urls):
            fetched = await self._fetch_subscription(url)
            if not fetched:
                continue
            fetched_parts.append(fetched)
            try:
                self._cache_file.parent.mkdir(parents=True, exist_ok=True)
                self._cache_file.write_text(fetched, encoding="utf-8")
            except OSError:
                pass

        text = "\n".join([*raw_lines, *fetched_parts])

        if "vless://" not in text:
            try:
                if self._cache_file.exists():
                    text = self._cache_file.read_text(
                        encoding="utf-8", errors="ignore"
                    )
                    self.log.info("VLESS: подписка недоступна, использую кэш")
            except OSError:
                pass

        return text

    async def _fetch_subscription(self, url: str) -> str:
        headers = {"User-Agent": "v2rayN/6.31"}
        timeout = aiohttp.ClientTimeout(total=20)
        try:
            async with aiohttp.ClientSession(timeout=timeout, headers=headers) as session:
                async with session.get(url) as response:
                    if response.status != 200:
                        self.log.warning(
                            "VLESS: подписка вернула HTTP %s", response.status
                        )
                        return ""
                    text = await response.text(errors="ignore")
                    if "vless://" not in text:
                        self.log.warning("VLESS: в подписке нет vless:// ссылок")
                        return ""
                    return text
        except (aiohttp.ClientError, asyncio.TimeoutError, OSError) as exc:
            self.log.warning("VLESS: не удалось скачать подписку: %s", exc)
            return ""

    # -- Xray-core ----------------------------------------------------------

    def _ensure_xray(self) -> Optional[str]:
        vless_cfg = self._section("vless")

        custom = str(vless_cfg.get("core_path") or "").strip()
        if custom:
            path = Path(custom)
            if not path.is_absolute():
                path = self.base_dir / path
            if path.exists():
                return str(path)

        bin_dir = self.base_dir / "bin" / "xray"
        exe_name = "xray.exe" if os.name == "nt" else "xray"
        exe = bin_dir / exe_name
        if exe.exists():
            return str(exe)

        if not bool(vless_cfg.get("auto_download", True)):
            return None

        url = self._xray_download_url()
        if not url:
            self.log.error(
                "VLESS: нет готового архива Xray для платформы %s/%s",
                sys.platform,
                platform.machine(),
            )
            return None

        try:
            bin_dir.mkdir(parents=True, exist_ok=True)
        except OSError as exc:
            self.log.error("VLESS: не удалось создать %s: %s", bin_dir, exc)
            return None

        archive = bin_dir / "xray-download.zip"
        self.log.info("VLESS: скачиваю Xray-core (%s)", url)
        try:
            request = urllib.request.Request(
                url, headers={"User-Agent": "KrimLogs/1.0"}
            )
            with urllib.request.urlopen(request, timeout=120) as response:
                with open(archive, "wb") as handle:
                    shutil.copyfileobj(response, handle)
            with zipfile.ZipFile(archive) as zip_file:
                zip_file.extractall(bin_dir)
            archive.unlink(missing_ok=True)
        except Exception as exc:
            self.log.error("VLESS: не удалось скачать/распаковать Xray: %s", exc)
            try:
                archive.unlink(missing_ok=True)
            except OSError:
                pass
            return None

        if not exe.exists():
            for candidate in bin_dir.rglob(exe_name):
                try:
                    shutil.move(str(candidate), str(exe))
                    break
                except OSError:
                    continue

        if not exe.exists():
            self.log.error("VLESS: в архиве не найден %s", exe_name)
            return None

        if os.name != "nt":
            try:
                exe.chmod(0o755)
            except OSError:
                pass

        self.log.info("VLESS: Xray-core готов: %s", exe)
        return str(exe)

    @staticmethod
    def _xray_download_url() -> str:
        machine = platform.machine().lower()
        base = "https://github.com/XTLS/Xray-core/releases/latest/download/"
        if os.name == "nt":
            return base + "Xray-windows-64.zip"
        if sys.platform == "darwin":
            if machine in ("arm64", "aarch64"):
                return base + "Xray-macos-arm64-v8a.zip"
            return base + "Xray-macos-64.zip"
        if sys.platform.startswith("linux"):
            if machine in ("arm64", "aarch64"):
                return base + "Xray-linux-arm64-v8a.zip"
            if machine in ("armv7l", "armv7"):
                return base + "Xray-linux-arm32-v7a.zip"
            return base + "Xray-linux-64.zip"
        return ""

    async def _start_proxy(self) -> Optional[str]:
        self._mode = "proxy"
        return await self._refresh_proxy_pool(keep_active=False)

    def _load_proxy_cache(self) -> list[str]:
        try:
            if self._proxy_cache_file.exists():
                data = json.loads(
                    self._proxy_cache_file.read_text(encoding="utf-8")
                )
                if isinstance(data, list):
                    return [
                        str(item).strip()
                        for item in data
                        if str(item).strip().startswith(("http://", "https://"))
                    ]
        except (OSError, ValueError):
            pass
        return []

    def _save_proxy_cache(self, proxies: list[str]) -> None:
        try:
            self._proxy_cache_file.parent.mkdir(parents=True, exist_ok=True)
            self._proxy_cache_file.write_text(
                json.dumps(proxies[: self._cache_limit], ensure_ascii=False, indent=2),
                encoding="utf-8",
            )
        except OSError:
            self.log.debug("Не удалось сохранить кэш прокси", exc_info=True)

    async def _refresh_proxy_pool(
        self, keep_active: bool, force_sources: bool = False
    ) -> Optional[str]:
        async with self._refresh_lock:
            proxy_cfg = self._section("proxy")
            sources = proxy_cfg.get("sources")
            if not isinstance(sources, (list, tuple)) or not sources:
                sources = DEFAULT_PROXY_SOURCES
            sources = [str(item) for item in sources if str(item).strip()]

            cached_hosts = [
                url.split("://", 1)[-1].strip("/")
                for url in self._load_proxy_cache()
            ]
            cached_hosts = [host for host in cached_hosts if host]

            headers = {"User-Agent": "Mozilla/5.0 (KrimLogs proxy scanner)"}
            timeout = aiohttp.ClientTimeout(total=25)
            connector = aiohttp.TCPConnector(
                limit=self._concurrency, force_close=True
            )
            results: list[tuple[float, str]] = []
            wanted = self._pool_size + 1

            async with aiohttp.ClientSession(
                timeout=timeout, headers=headers, connector=connector
            ) as session:
                if cached_hosts:
                    self.log.info(
                        "Прокси: проверяю кэш прошлых запусков (%d адресов)",
                        len(cached_hosts),
                    )
                    results.extend(await self._probe_all(session, cached_hosts))

                if force_sources or len(results) < wanted:
                    self.log.info(
                        "Прокси: скачиваю списки из %d источников", len(sources)
                    )
                    texts = await asyncio.gather(
                        *(self._fetch_text(session, source) for source in sources),
                        return_exceptions=True,
                    )
                    candidates: set[str] = set()
                    for text in texts:
                        if isinstance(text, str) and text:
                            candidates.update(self._parse_proxy_lines(text))

                    tested = set(cached_hosts)
                    fresh = [item for item in candidates if item not in tested]
                    random.shuffle(fresh)

                    if not fresh and not results:
                        self.log.error("Прокси: списки пусты или недоступны")
                        return None

                    batch = fresh[: self._max_candidates]
                    if batch:
                        self.log.info(
                            "Прокси: найдено %d адресов, проверяю HTTPS "
                            "(параллельно %d)",
                            len(batch),
                            self._concurrency,
                        )
                        results.extend(await self._probe_all(session, batch))

                    # Если рабочих меньше, чем «активный + запасные», добираем
                    # ещё одну порцию кандидатов из тех же списков.
                    if len(results) < wanted and len(fresh) > len(batch):
                        extra = fresh[
                            len(batch) : len(batch) + self._max_candidates
                        ]
                        if extra:
                            self.log.info(
                                "Прокси: рабочих %d, добираю ещё %d адресов",
                                len(results),
                                len(extra),
                            )
                            results.extend(await self._probe_all(session, extra))
                else:
                    self.log.info(
                        "Прокси: кэш дал %d рабочих адресов, списки не нужны",
                        len(results),
                    )

            if not results:
                self.log.error(
                    "Прокси: ни один адрес не открыл Discord по HTTPS. "
                    "Попробуйте режим vless или увеличьте max_candidates"
                )
                return None

            best: dict[str, float] = {}
            for latency, url in results:
                previous = best.get(url)
                if previous is None or latency < previous:
                    best[url] = latency
            merged = sorted(
                ((latency, url) for url, latency in best.items()),
                key=lambda item: item[0],
            )

            pool = [url for _latency, url in merged]
            if keep_active and self._proxy_url:
                active = self._proxy_url
                pool = [item for item in pool if item != active]
            else:
                active = pool.pop(0)

            self._proxy_url = active
            self._backups = pool[: self._pool_size]
            self._last_refresh = time.time()
            cache_urls = [
                active,
                *(url for _latency, url in merged if url != active),
            ]
            self._save_proxy_cache(cache_urls)

            best_latency = merged[0][0]
            self.log.info(
                "Прокси: активный %s (~%.0f мс), запасных %d",
                active,
                best_latency,
                len(self._backups),
            )
            if len(self._backups) < self._pool_size:
                self.log.warning(
                    "Прокси: найдено только %d запасных из %d — "
                    "бесплатные прокси нестабильны, при отказе будет новый поиск",
                    len(self._backups),
                    self._pool_size,
                )
            return active

    async def _fetch_text(self, session: aiohttp.ClientSession, url: str) -> str:
        try:
            async with session.get(
                url, timeout=aiohttp.ClientTimeout(total=25)
            ) as response:
                if response.status != 200:
                    return ""
                return await response.text(errors="ignore")
        except (aiohttp.ClientError, asyncio.TimeoutError, OSError):
            self.log.debug("Источник прокси недоступен: %s", url)
            return ""

    @staticmethod
    def _parse_proxy_lines(text: str) -> set[str]:
        found: set[str] = set()
        for line in text.splitlines():
            line = line.strip()
            if not line or line[0] in "#;/":
                continue
            host = ""
            port = ""
            if "://" in line:
                try:
                    parsed = urlsplit(line)
                    host = parsed.hostname or ""
                    port = str(parsed.port or "")
                except ValueError:
                    continue
            else:
                parts = line.split(":")
                if len(parts) < 2:
                    continue
                host, port = parts[0].strip(), parts[1].strip()
            if not host or not port:
                continue
            try:
                port_number = int(port)
            except ValueError:
                continue
            if not 0 < port_number < 65536:
                continue
            try:
                address = ipaddress.ip_address(host)
                if (
                    address.is_private
                    or address.is_loopback
                    or address.is_reserved
                    or address.is_multicast
                    or address.is_unspecified
                ):
                    continue
            except ValueError:
                # доменное имя — оставляем, проверит тест
                pass
            found.add(f"{host}:{port_number}")
        return found

    async def _probe_all(
        self,
        session: aiohttp.ClientSession,
        candidates: list[str],
        confirm: Optional[int] = None,
    ) -> list[tuple[float, str]]:
        semaphore = asyncio.Semaphore(self._concurrency)
        results: list[tuple[float, str]] = []

        async def worker(host_port: str) -> None:
            async with semaphore:
                proxy_url = f"http://{host_port}"
                latency = await self._test_proxy(
                    proxy_url, session, confirm=confirm
                )
                if latency is not None:
                    results.append((latency, proxy_url))

        await asyncio.gather(*(worker(item) for item in candidates))
        results.sort(key=lambda item: item[0])
        return results

    async def _proxy_health(self) -> None:
        if not self._proxy_url:
            url = await self._refresh_proxy_pool(keep_active=False)
            if url:
                self._apply(url, "restore")
            return

        if await self._test_proxy(self._proxy_url) is not None:
            refresh_minutes = max(
                1, self._int(self._section("proxy"), "refresh_minutes", 30)
            )
            if time.time() - self._last_refresh > refresh_minutes * 60:
                await self._refresh_proxy_pool(keep_active=True, force_sources=True)
            return

        self.log.warning("Прокси: активный %s не отвечает — ротация", self._proxy_url)
        while self._backups:
            candidate = self._backups.pop(0)
            if await self._test_proxy(candidate) is not None:
                self._apply(candidate, "rotate")
                return
            self.log.warning("Прокси: запасной %s тоже не отвечает", candidate)

        self.log.warning("Прокси: запасные закончились — ищу новые")
        url = await self._refresh_proxy_pool(keep_active=False)
        if url:
            self._apply(url, "rescan")

    # -- общий watchdog и применение ---------------------------------------

    async def _watchdog(self) -> None:
        while not self._stop.is_set():
            try:
                await asyncio.wait_for(self._stop.wait(), timeout=self._health_interval)
                break
            except asyncio.TimeoutError:
                pass
            try:
                if self._mode == "vless":
                    await self._vless_health()
                elif self._mode == "proxy":
                    await self._proxy_health()
            except asyncio.CancelledError:
                raise
            except Exception:
                self.log.exception("Ошибка проверки сетевого обхода")

    def _apply(self, url: Optional[str], reason: str) -> None:
        changed = url != self._proxy_url
        self._proxy_url = url
        if self._client is not None:
            try:
                self._client.http.proxy = url
            except Exception:
                self.log.exception("Не удалось обновить прокси в discord.py")
            if changed:
                self._schedule_reconnect()
        if changed:
            self.log.info("Сетевой обход переключён (%s): %s", reason, url)

    def _schedule_reconnect(self) -> None:
        client = self._client
        if client is None:
            return
        websocket = getattr(client, "ws", None)
        if websocket is None:
            return
        try:
            loop = asyncio.get_running_loop()
        except RuntimeError:
            return
        loop.create_task(self._close_websocket(websocket))

    async def _close_websocket(self, websocket: Any) -> None:
        try:
            if not getattr(websocket, "closed", False):
                await websocket.close(code=4000, message=b"netbypass rotation")
                self.log.info(
                    "WebSocket закрыт, discord.py переподключится через новый прокси"
                )
        except Exception:
            self.log.debug("Не удалось закрыть WebSocket", exc_info=True)
